        await stop().catch(err => logger.error({ err }, 'Erreur lors de l\'arrêt'));
        process.exit(0);
      });
 
      process.on('SIGINT', async () => {
        logger.info('SIGINT reçu, arrêt gracieux...');
        await continuousImprovement.stop();
        await stop().catch(err => logger.error({ err }, 'Erreur lors de l\'arrêt'));
        process.exit(0);
      });
    })
    .catch(err => {
      logger.fatal({ err }, 'Erreur fatale au démarrage');
      process.exit(1);
    });
